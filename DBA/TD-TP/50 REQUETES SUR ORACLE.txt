SELECT p.nomProduit, COUNT(DISTINCT co.idClient) AS nb_clients

JOIN commande co ON lc.idCom = co.idCom
GROUP BY p.nomProduit
ORDER BY nb_clients DESC
FETCH FIRST 1 ROWS ONLY;

SELECT c.*

    HAVING MIN(co.dateCom) >= TO_DATE('2023-01-01', 'YYYY-MM-DD')
       AND MIN(co.dateCom) <= TO_DATE('2023-12-31', 'YYYY-MM-DD')
);

9. Requêtes avancées
Trouver le client ayant passé la commande la plus ancienne :

SELECT c.nomClient, c.prenomClient
FROM client c
JOIN commande co ON c.idClient = co.idClient
WHERE co.dateCom = (SELECT MIN(dateCom) FROM commande);
Lister les commandes contenant le produit le moins commandé :

SELECT lc.idCom
FROM ligneCom lc
WHERE lc.idProduit = (
    SELECT idProduit
    FROM ligneCom
    GROUP BY idProduit
    ORDER BY SUM(qteCom) ASC
    FETCH FIRST 1 ROWS ONLY
);
Afficher les clients ayant passé au moins une commande pour chaque produit disponible :

SELECT c.idClient, c.nomClient, c.prenomClient
FROM client c
WHERE NOT EXISTS (
    SELECT 1
    FROM produit p
    WHERE NOT EXISTS (
        SELECT 1
        FROM commande co
        JOIN ligneCom lc ON co.idCom = lc.idCom
        WHERE lc.idProduit = p.idProduit AND co.idClient = c.idClient
    )
);
Calculer la moyenne des quantités commandées par produit :

SELECT p.nomProduit, AVG(lc.qteCom) AS avg_qte
FROM produit p
JOIN ligneCom lc ON p.idProduit = lc.idProduit
GROUP BY p.nomProduit;
Afficher les commandes où tous les produits commandés ont une quantité supérieure à 100 :

SELECT idCom
FROM ligneCom
GROUP BY idCom
HAVING MIN(qteCom) > 100;
Trouver les produits commandés par exactement deux clients différents :

SELECT p.nomProduit
FROM produit p
JOIN ligneCom lc ON p.idProduit = lc.idProduit
JOIN commande co ON lc.idCom = co.idCom
GROUP BY p.nomProduit
HAVING COUNT(DISTINCT co.idClient) = 2;
Lister les produits qui représentent au moins 50% des quantités commandées dans une commande :

SELECT p.nomProduit, lc.idCom
FROM produit p
JOIN ligneCom lc ON p.idProduit = lc.idProduit
GROUP BY p.nomProduit, lc.idCom
HAVING SUM(lc.qteCom) >= 0.5 * (
    SELECT SUM(qteCom)
    FROM ligneCom
    WHERE idCom = lc.idCom
);
Trouver les clients ayant passé exactement une commande avec plusieurs lignes :

SELECT c.nomClient, c.prenomClient
FROM client c
JOIN commande co ON c.idClient = co.idClient
WHERE co.idCom IN (
    SELECT idCom
    FROM ligneCom
    GROUP BY idCom
    HAVING COUNT(*) > 1
)
AND c.idClient NOT IN (
    SELECT idClient
    FROM commande
    WHERE idCom NOT IN (
        SELECT idCom
        FROM ligneCom
        GROUP BY idCom
        HAVING COUNT(*) > 1
    )
);
