### 2. Code des servlets

#### Servlet `QCM.java`

```java
import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet("/QCM")
public class QCM extends HttpServlet {
    protected void processRequest(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String epreuve = request.getParameter("epreuve");
        QuestionReponses qr = new QuestionReponses();
        
        if (epreuve.equals("Algèbre")) {
            response.setStatus(HttpServletResponse.SC_NOT_IMPLEMENTED);
            response.getWriter().write("Page en construction");
            return;
        }

        int nombreQuestions = qr.getNombreQuestion(epreuve);
        request.setAttribute("epreuve", epreuve);
        request.setAttribute("nombreQuestions", nombreQuestions);
        request.setAttribute("questionReponses", qr);
        request.getRequestDispatcher("/qcm.jsp").forward(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }
}
```

#### Servlet `ResultatsQCM.java`

```java
import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet("/ResultatsQCM")
public class ResultatsQCM extends HttpServlet {
    protected void processRequest(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        String epreuve = request.getParameter("epreuve");
        QuestionReponses qr = new QuestionReponses();
        int nombreQuestions = qr.getNombreQuestion(epreuve);
        StringBuilder resultats = new StringBuilder();

        for (int i = 0; i < nombreQuestions; i++) {
            String reponseUtilisateur = request.getParameter("reponse" + i);
            String bonneReponse = qr.getReponse(epreuve, i);
            resultats.append("Question: ")
                      .append(qr.getQuestion(epreuve, i))
                      .append("<br>Votre réponse: ")
                      .append(reponseUtilisateur)
                      .append("<br>Bonne réponse: ")
                      .append(bonneReponse)
                      .append("<br><br>");
        }

        request.setAttribute("resultats", resultats.toString());
        request.getRequestDispatcher("/resultats.jsp").forward(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        processRequest(request, response);
    }
}
```
